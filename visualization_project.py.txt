# PYTHON DATA VISUALIZATION PROJECT
# Tools Used:
# - Python
# - NumPy
# - Matplotlib

import matplotlib.pyplot as plt
import numpy as np


# -----------------------------
# 1. LINE CHART
# -----------------------------
months = np.array([1, 2, 3, 4, 5])
sales = np.array([10, 20, 30, 25, 35])

plt.figure(figsize=(6, 4))
plt.plot(months, sales, marker="o", color="blue")
plt.title("Monthly Sales")
plt.xlabel("Months")
plt.ylabel("Sales")
plt.grid()
plt.show()


# -----------------------------
# 2. BAR CHART
# -----------------------------
students = np.array(["Ram", "Ravi", "Anu", "Kiran"])
marks = np.array([80, 70, 90, 85])

plt.figure(figsize=(6, 4))
plt.bar(students, marks, color="green")
plt.title("Student Marks")
plt.xlabel("Students")
plt.ylabel("Marks")
plt.grid(axis="y")
plt.show()


# -----------------------------
# 3. PIE CHART
# -----------------------------
expenses = np.array([40, 30, 20, 10])
labels = np.array(["Food", "Rent", "Travel", "Other"])

plt.figure(figsize=(6, 6))
plt.pie(expenses, labels=labels, autopct="%1.1f%%")
plt.title("Monthly Expenses")
plt.show()


# -----------------------------
# 4. HISTOGRAM
# -----------------------------
data = np.array([10, 20, 20, 30, 40, 40, 50, 60, 70])

plt.figure(figsize=(6, 4))
plt.hist(data, bins=5)
plt.title("Histogram Example")
plt.xlabel("Values")
plt.ylabel("Frequency")
plt.grid()
plt.show()


# -----------------------------
# 5. SCATTER PLOT
# -----------------------------
hours = np.array([1, 2, 3, 4, 5])
study_marks = np.array([50, 55, 65, 70, 80])

plt.figure(figsize=(6, 4))
plt.scatter(hours, study_marks, color="red")
plt.title("Study Hours vs Marks")
plt.xlabel("Study Hours")
plt.ylabel("Marks")
plt.grid()
plt.show()


# -----------------------------
# 6. MULTIPLE LINE CHART
# -----------------------------
months = np.array([1, 2, 3, 4])

sales = np.array([10, 20, 30, 40])
profit = np.array([5, 10, 15, 20])

plt.figure(figsize=(6, 4))
plt.plot(months, sales, marker="o", label="Sales")
plt.plot(months, profit, marker="o", label="Profit")

plt.title("Sales and Profit Analysis")
plt.xlabel("Months")
plt.ylabel("Amount")
plt.legend()
plt.grid()
plt.show()


# -----------------------------
# 7. SUBPLOTS
# -----------------------------
plt.figure(figsize=(10, 8))

# Line Chart
plt.subplot(2, 2, 1)
plt.plot(months, sales, marker="o")
plt.title("Sales Chart")

# Bar Chart
plt.subplot(2, 2, 2)
plt.bar(students, marks)
plt.title("Marks Chart")

# Pie Chart
plt.subplot(2, 2, 3)
plt.pie(expenses, labels=labels, autopct="%1.1f%%")
plt.title("Expense Chart")

# Scatter Plot
plt.subplot(2, 2, 4)
plt.scatter(hours, study_marks)
plt.title("Study Analysis")

plt.tight_layout()
plt.show()
